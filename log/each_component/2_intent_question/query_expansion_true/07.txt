Complete EOL example (runnable in an Epsilon script with the YAML EMC model available):

```eol
// Set root to a mapping node
YAMLDoc.setRootAsMap();
var root = YAMLDoc.getRoot();

// server: {host: 0.0.0.0, port: 8443, protocol: https, maxConnections: 500}
var serverNode = new MappingNode("server");
var hostNode = new ScalarNode("host", "0.0.0.0");
var portNode = new ScalarNode("port", 8443);
var protocolNode = new ScalarNode("protocol", "https");
var maxConnectionsNode = new ScalarNode("maxConnections", 500);
serverNode.value.appendNode(hostNode);
serverNode.value.appendNode(portNode);
serverNode.value.appendNode(protocolNode);
serverNode.value.appendNode(maxConnectionsNode);
root.value.appendNode(serverNode);

// database: {engine: postgresql, host: db01.internal, port: 5432, poolSize: 20}
var databaseNode = new MappingNode("database");
var engineNode = new ScalarNode("engine", "postgresql");
var dbHostNode = new ScalarNode("host", "db01.internal");
var dbPortNode = new ScalarNode("port", 5432);
var poolSizeNode = new ScalarNode("poolSize", 20);
databaseNode.value.appendNode(engineNode);
databaseNode.value.appendNode(dbHostNode);
databaseNode.value.appendNode(dbPortNode);
databaseNode.value.appendNode(poolSizeNode);
root.value.appendNode(databaseNode);

// allowedOrigins: [https://app.example.com, https://admin.example.com, https://staging.example.com]
var allowedOriginsNode = new ListNode("allowedOrigins", 3);
allowedOriginsNode.value = Sequence{
  "https://app.example.com",
  "https://admin.example.com",
  "https://staging.example.com"
};
root.value.appendNode(allowedOriginsNode);

// rateLimits: [{tier: free, requestsPerMinute: 60}, {tier: pro, requestsPerMinute: 600}]
var rateLimitsNode = new ListNode("rateLimits", 2);

// first rate limit entry
var tier1 = new ScalarNode("tier", "free");
var rpm1 = new ScalarNode("requestsPerMinute", 60);
rateLimitsNode.value.at(0).appendNode(tier1);
rateLimitsNode.value.at(0).appendNode(rpm1);

// second rate limit entry
var tier2 = new ScalarNode("tier", "pro");
var rpm2 = new ScalarNode("requestsPerMinute", 600);
rateLimitsNode.value.at(1).appendNode(tier2);
rateLimitsNode.value.at(1).appendNode(rpm2);

root.value.appendNode(rateLimitsNode);
```

Notes:
- The root must be a mapping node for this YAML (so we call YAMLDoc.setRootAsMap()).
- This code follows the patterns shown in the YAML EMC examples: use MappingNode, ScalarNode, ListNode, appendNode, and Sequence as demonstrated.
